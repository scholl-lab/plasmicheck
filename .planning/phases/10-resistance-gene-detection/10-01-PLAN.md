- `EXCLUDE_PATTERNS`: His-tag/His_tag/6His, FLAG, Myc, V5, HA (affinity tags to skip)
- Combine as `DEFAULT_PATTERNS = BACTERIAL + MAMMALIAN + REPORTER`

```python
@dataclass
class DetectedGene:
    """A resistance/reporter gene detected from GenBank annotations."""
    name: str           # Verbatim GenBank label (display name)
    sanitized_name: str # For TSV row keys (spaces/special -> underscores)
    start: int          # 0-based inclusive
    end: int            # 0-based inclusive
    region: str         # "backbone" or "insert" (or "unknown" if no insert region)
    qualifier_source: str  # Which qualifier matched ("label", "product", "gene")

def sanitize_gene_name(name: str) -> str:
    """Replace non-alphanumeric chars (except _ and -) with underscores."""

def _matches_resistance_gene(text: str, patterns: list[str], exclude_patterns: list[str]) -> bool:
    """Check if text matches any pattern and is not excluded."""

def _identify_gene_name(qualifiers: dict[str, list[str]], patterns: list[str], exclude_patterns: list[str]) -> tuple[str, str] | None:
    """Return (gene_name, qualifier_source) or None. Check /label, /product, /gene in priority order."""

def _gene_overlaps_insert(gene_start: int, gene_end: int, insert_region: tuple[int, int] | None) -> bool:
    """Check if gene overlaps insert region. Returns False if insert_region is None."""

def detect_resistance_genes(
    genbank_file: str,
    insert_region: tuple[int, int] | None = None,
    custom_patterns: list[str] | None = None,
) -> list[DetectedGene]:
    """Parse GenBank CDS features, match against patterns, exclude insert-overlapping genes.

    Args:
        genbank_file: Path to .gb/.gbk file
        insert_region: (start, end) inclusive boundaries from cDNA_positions.txt, or None
        custom_patterns: Additional regex patterns from config.json (appended to defaults)

    Returns:
        List of DetectedGene objects for matched genes outside the insert region.
        Empty list if no genes match or file cannot be parsed.
    """
```

- Parse GenBank using `Bio.SeqIO.parse(file, "genbank")`
- Iterate CDS features only (`feature.type == "CDS"`)
- For each CDS, extract qualifiers with `.get(key, [""])[0]` pattern (qualifiers are lists)
- Match against patterns (defaults + custom if provided)
- Use `feature.location.start` (0-based) and `feature.location.end - 1` (convert to inclusive) for coordinates
- Handle `CompoundLocation` (origin-crossing features): use `feature.location.parts`, take min start and max end across parts
- Classify region: if insert_region provided, check overlap; genes overlapping insert are EXCLUDED (logged at debug level); genes not overlapping are "backbone"; if no insert_region, region is "unknown"
- Return list of DetectedGene, sorted by start position
- Wrap entire function in try/except for BioPython parse errors, log warning and return empty list
